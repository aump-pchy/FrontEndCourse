const handleAdd = async () => { // validation if (!form.name.trim() || !form.job.trim()) { alert("กรุณากรอกชื่อและตำแหน่ง") return } try { setAdding(true) const res = await fetch(`${BASE}/users`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(form) }) const created = await res.json() // reqres ตอบ: { id, name, job, createdAt } // ต้องแปลงให้ตรงกับ UserCard props const newUser = { id: created.id, first_name: form.name, last_name: "", email: `${form.name.toLowerCase()}@reqres.in`, avatar: "https://reqres.in/img/faces/10-image.jpg" } setUsers(prev => [...prev, newUser]) // เพิ่มลง state setForm({ name: "", job: "" }) // reset form } catch (e) { alert("เพิ่มไม่สำเร็จ: " + e.message) } finally { setAdding(false) } }

❓ คำถามที่นักเรียนมักถาม
❓ ทำไม json.data ไม่ใช่ json ตรงๆ?
reqres.in ออกแบบ response ให้มีข้อมูลเพิ่มเติมด้วย เช่น page, total, per_page จึงห่อ users array ไว้ใน key ชื่อ "data" — เปิด DevTools Network ดู response structure จะเห็นชัด
❓ ทำไมต้อง [...prev, newUser] ไม่ใช้ setUsers([...users, newUser])?
ทั้งสองแบบทำงานได้ในกรณีส่วนใหญ่ แต่ prev => แบบ callback มั่นใจได้ว่าได้ state ล่าสุดเสมอ ในกรณีที่มีการ update หลาย state พร้อมกัน prev => จะปลอดภัยกว่า
❓ reqres.in DELETE แล้ว user กลับมาเมื่อ refresh ไหม?
ใช่ — reqres.in ไม่บันทึกข้อมูลจริงๆ แค่ simulate response ดังนั้น refresh ใหม่ user กลับมา นั่นเป็นเหตุผลที่เราต้อง filter state ใน React เพื่อให้ UI ดูถูกต้องในระหว่างใช้งาน
❓ ทำไม DELETE ถึงไม่มี body?
HTTP spec กำหนดว่า DELETE ระบุ resource ที่จะลบผ่าน URL แล้ว (เช่น /api/users/2) ไม่ต้องส่งข้อมูลเพิ่มเติม — เหมือนบอกว่า "ลบอันนี้" ชี้นิ้วไปที่ URL เลย
❓ [page] ใน dependency array หมายความว่าอะไร?
บอก React ว่า "รัน useEffect ใหม่ทุกครั้งที่ page เปลี่ยน" — เมื่อกด "หน้า 2" state page เปลี่ยนเป็น 2 → React เห็นว่า page เปลี่ยน → รัน useEffect อีกครั้ง → fetch /api/users?page=2 ใหม่
❓ ถ้าต้องการ PUT แก้ไข User ต้องทำยังไง?
สร้าง state editUser เก็บ user ที่เลือก → เปิด modal/form → กรอกข้อมูล → PUT ไปที่ /api/users/{'{'}id{'}'} พร้อม body → อัปเดต state ด้วย setUsers(prev => prev.map(u => u.id === id ? updated : u)) — เป็น Bonus ของชิ้นงาน
