💻 App.jsx — โครงสร้างขั้นต่ำ
import { useState } from 'react' import ItemCard from './components/ItemCard' // ← W9: array of objects const data = [ { id: 1, name: "...", category: "...", emoji: "...", price: 0 }, // ... ≥ 6 items ] const App = () => { const [query, setQuery] = useState("") // ← W10 const [category, setCategory] = useState("ทั้งหมด") // ← W10 // ← W11: filter + search const filtered = data .filter(i => category === "ทั้งหมด" || i.category === category) .filter(i => i.name.includes(query)) return ( <div className="p-6"> {/* ← W10: controlled input */} <input value={query} onChange={e => setQuery(e.target.value)} /> {/* ← W11: .map() */} <div className="grid grid-cols-1 md:grid-cols-3 gap-4"> {filtered.map(item => ( <ItemCard key={item.id} {...item} /> // ← W8, W9 ))} </div> </div> ) }
