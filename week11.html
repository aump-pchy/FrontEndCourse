เซลที่ 4 — ตรวจจับบุคคลทีละเฟรม
import cv2

# ─── เปิดไฟล์วิดีโอที่อัปโหลด ───
cap = cv2.VideoCapture(video_filename)

# ─── อ่านข้อมูลของวิดีโอต้นฉบับ ───
fps    = int(cap.get(cv2.CAP_PROP_FPS))          # ความเร็วเฟรม (เฟรม/วินาที)
width  = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))  # ความกว้างวิดีโอ
height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT)) # ความสูงวิดีโอ
total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))

print(f"วิดีโอต้นฉบับ: {width}x{height}, {fps} FPS, {total_frames} เฟรม")

# ─── เตรียมไฟล์วิดีโอผลลัพธ์ ───
fourcc = cv2.VideoWriter_fourcc(*'mp4v')
out = cv2.VideoWriter('output_result.mp4', fourcc, fps, (width, height))

# ─── ตัวนับจำนวนบุคคลสูงสุดที่เจอในเฟรมเดียว ───
max_person_count = 0
frame_number = 0

# ─── วนลูปอ่านวิดีโอทีละเฟรมจนกว่าจะหมด ───
while cap.isOpened():
    ret, frame = cap.read()   # ret=True ถ้าอ่านเฟรมได้สำเร็จ
    if not ret:
        break    # วิดีโอหมดแล้ว ออกจากลูป

    frame_number += 1

    # ─── ส่งเฟรมนี้ให้ YOLO ตรวจจับ กรองเฉพาะ class 0 (person) ───
    results = model(frame, classes=[0], conf=0.4, verbose=False)
    result = results[0]

    # ─── นับจำนวนคนที่เจอในเฟรมนี้ ───
    person_count = len(result.boxes)
    if person_count > max_person_count:
        max_person_count = person_count

    # ─── วาดกรอบ + label บนเฟรม ───
    annotated_frame = result.plot()

    # ─── เขียนตัวเลขจำนวนคนที่มุมซ้ายบนของเฟรม ───
    cv2.putText(annotated_frame, f"Persons: {person_count}",
                (20, 40), cv2.FONT_HERSHEY_SIMPLEX,
                1, (0, 255, 0), 2)

    # ─── บันทึกเฟรมนี้ลงในวิดีโอผลลัพธ์ ───
    out.write(annotated_frame)

    # ─── แสดงความคืบหน้าทุก 30 เฟรม ───
    if frame_number % 30 == 0:
        print(f"ประมวลผลแล้ว {frame_number}/{total_frames} เฟรม")

# ─── ปิดไฟล์วิดีโอทั้งสอง ───
cap.release()
out.release()

print(f"\n✅ เสร็จสิ้น! ประมวลผลทั้งหมด {frame_number} เฟรม")
print(f"👤 พบจำนวนบุคคลสูงสุดในเฟรมเดียว: {max_person_count} คน")

        async function startStream() {

            video.srcObject = stream;
            await video.play();
            captureCanvas = document.createElement('canvas');
            captureCanvas.width = 640;
            captureCanvas.height = 480;
            return 'started';
        }

        async function grabFrame() {
            var ctx = captureCanvas.getContext('2d');
            ctx.drawImage(video, 0, 0, 640, 480);
            return captureCanvas.toDataURL('image/jpeg', 0.8);
        }

        window.startStream = startStream;
        window.grabFrame = grabFrame;
        window.removeDom = removeDom;
    ''')
    display(js)
    eval_js('startStream()')
    print("✅ กล้องเริ่มทำงานแล้ว — เบราว์เซอร์จะขอสิทธิ์เข้าถึงกล้อง กด Allow")

# ─── ฟังก์ชันดึงภาพ 1 เฟรมจากกล้อง แปลงเป็น OpenCV image ───
def get_webcam_frame():
    data_url = eval_js('grabFrame()')
    header, encoded = data_url.split(',', 1)
    img_bytes = b64decode(encoded)
    img_array = np.frombuffer(img_bytes, dtype=np.uint8)
    frame = cv2.imdecode(img_array, cv2.IMREAD_COLOR)
    return frame

# ─── ฟังก์ชันปิดกล้อง ───
def stop_webcam_stream():
    eval_js('removeDom()')
    print("ปิดกล้องแล้ว")

print("ตัวช่วยกล้องพร้อมใช้งาน — ไปทำเซล B ต่อได้เลย")

8.3 สร้างเซลใหม่ (เซล B) แล้วพิมพ์โค้ดนี้เพื่อเริ่มตรวจจับจากกล้องจริง
เซล B — ตรวจจับบุคคลจากเว็บแคม Real-time
import matplotlib.pyplot as plt
from IPython.display import clear_output

# ─── เปิดกล้อง (เบราว์เซอร์จะขอสิทธิ์ — กด Allow) ───
start_webcam_stream()

# ─── กำหนดจำนวนเฟรมที่จะจับ (ปรับได้ตามต้องการ) ───
num_frames = 30   # จับ 30 เฟรม แล้วหยุดอัตโนมัติ

for i in range(num_frames):
    # ─── ดึง 1 เฟรมจากกล้อง ───
    frame = get_webcam_frame()

    # ─── ส่งให้ YOLO ตรวจจับเฉพาะคน ───
    results = model(frame, classes=[0], conf=0.4, verbose=False)
    annotated = results[0].plot()
    person_count = len(results[0].boxes)

    # ─── แสดงผลแบบ real-time (เขียนทับภาพเดิมทุกเฟรม) ───
    clear_output(wait=True)
    annotated_rgb = cv2.cvtColor(annotated, cv2.COLOR_BGR2RGB)
    plt.figure(figsize=(8, 6))
    plt.imshow(annotated_rgb)
    plt.axis('off')
    plt.title(f'Frame {i+1}/{num_frames} — Persons: {person_count}')
    plt.show()

# ─── ปิดกล้องเมื่อจับครบตามจำนวนแล้ว ───
stop_webcam_stream()
print("✅ จบการตรวจจับแบบ real-time")
