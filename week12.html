// เพิ่มใน App.jsx // State สำหรับ Form (Week 10) const [form, setForm] = useState({ name: "", job: "" }) const [adding, setAdding] = useState(false) // POST Function (Week 12) const handleAdd = async () => { if (!form.name || !form.job) return // validation try { setAdding(true) const res = await fetch(`${BASE}/users`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(form) }) const created = await res.json() // reqres ตอบ: { id, name, job, createdAt } // เพิ่มลง state (Week 10) setUsers(prev => [...prev, { id: created.id, first_name: form.name, last_name: "", email: `${form.name.toLowerCase()}@reqres.in`, avatar: "https://reqres.in/img/faces/10-image.jpg" }]) setForm({ name: "", job: "" }) // reset form } catch (e) { setError(e.message) } finally { setAdding(false) } } // JSX Form <div className="bg-blue-50 border border-blue-200 rounded-xl p-4 mb-6"> <h2 className="text-blue-700 font-bold mb-3">➕ เพิ่ม User ใหม่</h2> <div className="flex gap-3 flex-wrap"> <input value={form.name} onChange={e => setForm({...form, name: e.target.value})} placeholder="ชื่อ" className="border rounded-lg px-3 py-2 flex-1" /> <input value={form.job} onChange={e => setForm({...form, job: e.target.value})} placeholder="ตำแหน่ง" className="border rounded-lg px-3 py-2 flex-1" /> <button onClick={handleAdd} disabled={adding} className="bg-blue-600 text-white px-6 py-2 rounded-lg"> {adding ?
